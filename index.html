(() => { "use strict"; // ---------- Setup ---------- const canvas = document.getElementById("game"); const ctx = canvas.getContext("2d"); const W = canvas.width; const H = canvas.height; const menu = document.getElementById("menu"); const bestMenu = document.getElementById("bestMenu"); const finalScore = document.getElementById("finalScore"); const newBest = document.getElementById("newBest"); const IMAGE_PATHS = { player: "assets/player.svg", lantern: "assets/lantern.svg", hazard: "assets/hazard.svg", }; const images = {}; const COLORS = { lantern: "#ffc857", hazard: "#7a52a0", player: "#e8e1d0", text: "#f3efe4" }; // ---------- Storage (safe) ---------- function loadBest() { try { return parseInt(localStorage.getItem("mothrun-best"), 10) || 0; } } // ---------- Update ---------- function update(dt) { // background stars always drift for (const s of stars) { s.y += s.v * dt; if (s.y > H) { s.y = -2; s.x = Math.random() * W; } } for (const p of particles) { p.x += p.vx * dt; sfx.hit(); if (state.lives <= 0) { endGame(); break; } } else if (h.y > H + 40) { h.dead = true; state.score += 1; // small reward for surviving } } hazards = hazards.filter((h) => !h.dead); } // ---------- Draw ---------- function drawSprite(img, x, y, size, fallbackColor, rotation = 0, alpha = 1) { ctx.save(); ctx.beginPath(); ctx.arc(0, 0, size / 2.4, 0, Math.PI * 2); ctx.fill(); } ctx.restore(); } function draw() { ctx.save(); if (state.shake > 0) ctx.translate(rand(-4, 4), rand(-4, 4)); // Background const grad = ctx.createLinearGradient(0, 0, 0, H); grad.addColorStop(0, "#0b1522"); grad.addColorStop(1, "#1b3550"); ctx.fillStyle = grad; ctx.fillRect(0, 0, W, H); ctx.fillStyle = "#cfe0f2"; for (const s of stars) { ctx.globalAlpha = 0.25 + s.s / 3; ctx.fillRect(s.x, s.y, s.s, s.s); } ctx.globalAlpha = 1; // Lanterns with glow for (const l of lanterns) { const g = ctx.createRadialGradient(l.x, l.y, 4, l.x, l.y, 46); g.addColorStop(0, "rgba(255,200,87,0.35)"); g.addColorStop(1, "rgba(255,200,87,0)"); ctx.fillStyle = g; ctx.fillRect(l.x - 46, l.y - 46, 92, 92); drawSprite(images.lantern, l.x, l.y, 44, COLORS.lantern); } for (const h of hazards) drawSprite(images.hazard, h.x, h.y, 54, COLORS.hazard, h.rot); // Player (blinks while invulnerable) if (state.mode !== "over") { const blink = state.invuln > 0 && Math.floor(state.invuln * 12) % 2 === 0; const bob = Math.sin(performance.now() / 120) * 2; drawSprite(images.player, player.x, player.y + bob, 56, COLORS.player, 0, blink ? 0.35 : 1); } // Particles for (const p of particles) { ctx.globalAlpha = Math.max(0, p.life / p.max); ctx.fillStyle = p.color; ctx.fillRect(p.x - 2, p.y - 2, 4, 4); } ctx.globalAlpha = 1; ctx.restore(); // HUD if (state.mode === "playing") { ctx.fillStyle = COLORS.text; for (let i = 0; i < 3; i++) { ctx.globalAlpha = i < state.lives ? 1 : 0.2; drawSprite(images.lantern, W - 26 - i * 28, 46, 22, COLORS.lantern); } ctx.globalAlpha = 1; } } // ---------- Loop ---------- let last = performance.now(); function frame(now) { const dt = Math.min(0.05, (now - last) / 1000); last = now; update(dt); draw(); requestAnimationFrame(frame); } // ---------- Boot: load images, then start loop ---------- function loadImages() { const jobs = Object.entries(IMAGE_PATHS).map(([name, src]) => new Promise((resolve) => { const img = new Image();
